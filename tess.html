# Tess AI — Полный проект (Frontend + Backend) Проект содержит минимальный, но **рабочий** стек: фронтенд (React + Vite + Tailwind-ready) и бэкенд (Node + Express), который проксирует запросы на выбранную модель (OpenAI или Hugging Face). В проекте есть demo-mode — интерфейс будет работать без ключей. --- ## Структура (файлы в репозитории) ``` / (repo root) ├─ package.json ├─ server.js ├─ .env.example ├─ README.md ├─ index.html ├─ vite.config.js ├─ tailwind.config.cjs (опционально) └─ src/ ├─ main.jsx └─ App.jsx ``` --- ## package.json ```json { "name": "tess-ai", "version": "0.1.0", "private": true, "scripts": { "dev": "vite", "build": "vite build", "preview": "vite preview --port 4173", "start": "node server.js" }, "dependencies": { "cors": "^2.8.5", "dotenv": "^16.0.0", "express": "^4.18.2", "node-fetch": "^3.3.1" }, "devDependencies": { "vite": "^5.0.0", "react": "^18.2.0", "react-dom": "^18.2.0" } } ``` > Примечание: Tailwind в `devDependencies` не включён — он опционален. Be direct, concise when asked for short answers, otherwise provide clear code examples. If the user sends code, respond with corrected/annotated code blocks and short explanation. Always label language when returning code blocks.`; } app.post('/api/ai', async (req, res) => { try { const ip = req.ip || req.connection.remoteAddress; if (!checkRateLimit(ip)) { return res.status(429).json({ error: 'Rate limit exceeded' }); } const { messages, user_input, persona } = req.body || {}; if (DEMO) { // quick demo reply return res.json({ reply: `Demo Tess reply for: ${user_input}` }); } // Build final prompt/messages const baseMessages = [ { role: 'system', content: systemPrompt() }, ]; // Append historical messages if any (but limit size) if (Array.isArray(messages)) { const tail = messages.slice(-8).map(m => ({ role: m.role, content: m.content })); baseMessages.push(...tail); } baseMessages.push({ role: 'user', content: user_input }); // Prefer OpenAI if API key present if (OPENAI_KEY) { const payload = { model: 'gpt-4o-mini', messages: baseMessages, temperature: 0.15, max_tokens: 1024, }; const resp = await fetch('https://api.openai.com/v1/chat/completions', { method: 'POST', headers: { 'Content-Type': 'application/json', 'Authorization': `Bearer ${OPENAI_KEY}` }, body: JSON.stringify(payload) }); if (!resp.ok) { const txt = await resp.text(); return res.status(502).json({ error: `OpenAI error: ${resp.status}`, detail: txt }); } const data = await resp.json(); const reply = data?.choices?.[0]?.message?.content || JSON.stringify(data); return res.json({ reply }); } // Else try Hugging Face Inference if (HF_KEY && HF_MODEL) { // Compose prompt const prompt = baseMessages.map(m => `${m.role}: ${m.content}`).join(' ') + ' assistant:'; const resp = await fetch(`https://api-inference.huggingface.co/models/${HF_MODEL}`, { method: 'POST', headers: { Authorization: `Bearer ${HF_KEY}`, 'Content-Type': 'application/json' }, body: JSON.stringify({ inputs: prompt, parameters: { max_new_tokens: 512, temperature: 0.15 } }) }); if (!resp.ok) { const txt = await resp.text(); return res.status(502).json({ error: `HF error: ${resp.status}`, detail: txt }); } const data = await resp.json(); // HF may respond with { generated_text } or array; try to extract let reply = ''; if (Array.isArray(data) && data[0] && data[0].generated_text) reply = data[0].generated_text; else if (data.generated_text) reply = data.generated_text; else if (data[0] && typeof data[0] === 'string') reply = data[0]; else reply = JSON.stringify(data); return res.json({ reply }); } // If no provider configured return res.json({ reply: 'Server not configured with OpenAI or Hugging Face keys. Set OPENAI_API_KEY or HF_API_KEY+HF_MODEL, or enable DEMO_MODE.' }); } catch (e) { console.error(e); return res.status(500).json({ error: e.message }); } }); app.listen(PORT, () => console.log(`Tess AI server running on http://localhost:${PORT}`)); ``` --- ## index.html ```html Tess AI — Femfur
